វិទ្យាស្ថានបច្ចេកវិទ្យាកំពង់ស្ពឺ

ការវាយតម្លៃគុណភាព & ហេដ្ឋារចនាសម្ព័ន្ធ (៥០ សំណួរ)

សិស្ស-និស្សិត
ផ្នែកទី ១៖ ព័ត៌មានទូទៅ ជំហាន ១/១០

I. ព័ត៌មានទូទៅ

សូមបំពេញព័ត៌មានមូលដ្ឋានខាងក្រោមដើម្បីចាប់ផ្តើម៖

II. ហេដ្ឋារចនាសម្ព័ន្ធបរិស្ថានទូទៅ និងសន្តិសុខ

តើប្អូនពេញចិត្តចំពោះហេដ្ឋារចនាសម្ព័ន្ធបរិស្ថានទូទៅកម្រិតណា?

III. ហេដ្ឋារចនាសម្ព័ន្ធអគារសិក្សា A (ការបង្រៀន & រៀន)

តើប្អូនពេញចិត្តចំពោះហេដ្ឋារចនាសម្ព័ន្ធអគារសិក្សា A កម្រិតណា?

IV. ហេដ្ឋារចនាសម្ព័ន្ធរោងជាង បន្ទប់ពិសោធន៍ & កសិដ្ឋាន

តើប្អូនពេញចិត្តចំពោះហេដ្ឋារចនាសម្ព័ន្ធអនុវត្តជាក់ស្តែងកម្រិតណា?

V. ហេដ្ឋារចនាសម្ព័ន្ធអគាររដ្ឋបាល & សេវាគាំទ្រ

តើប្អូនពេញចិត្តចំពោះអគាររដ្ឋបាល និងសេវាកម្មកម្រិតណា? (ញែកដាច់ដោយឡែក)

VI. ហេដ្ឋារចនាសម្ព័ន្ធអន្តេវាសិកដ្ឋានទូទៅ (អគារថ្មី & ABCD)

តើប្អូនពេញចិត្តចំពោះអន្តេវាសិកដ្ឋានកម្រិតណា?

VII. ហេដ្ឋារចនាសម្ព័ន្ធអគារស្នាក់នៅសូឡា (អគារស្នាក់នៅ & ថាមពល)

តើប្អូនពេញចិត្តចំពោះអគារស្នាក់នៅសូឡាកម្រិតណា?

VIII. ហេដ្ឋារចនាសម្ព័ន្ធបច្ចេកវិទ្យា និងប្រព័ន្ធ Wi-Fi

តើប្អូនពេញចិត្តចំពោះប្រព័ន្ធបច្ចេកវិទ្យា និង Wi-Fi កម្រិតណា?

IX. ហេដ្ឋារចនាសម្ព័ន្ធសុខុមាលភាព អាហារដ្ឋាន & កីឡា

តើប្អូនពេញចិត្តចំពោះសុខុមាលភាព និងជីវភាពរស់នៅកម្រិតណា?

X. ការវាយតម្លៃយុទ្ធសាស្ត្រ & មតិយោបល់

(០ = មិនណែនាំទាល់តែសោះ ដល់ ១០ = ណែនាំខ្លាំងបំផុត)

(សូមជ្រើសរើស ៣ ចំណុចសំខាន់បំផុត)
